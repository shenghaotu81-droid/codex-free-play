cmake_minimum_required(VERSION 3.16)
project(exchange_campus_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CAMPUS_SOURCE)
    message(FATAL_ERROR "Set CAMPUS_SOURCE to an extracted, pinned source snapshot")
endif()
set(VISION_ROOT "${CAMPUS_SOURCE}/src/tushenghao")
find_package(OpenCV 4.6 REQUIRED COMPONENTS core imgproc calib3d video videoio imgcodecs)
foreach(part types config detection pose tracking)
    add_subdirectory("${VISION_ROOT}/src/${part}" "campus/${part}")
endforeach()
add_executable(campus_bridge bridge.cpp)
target_link_libraries(campus_bridge PRIVATE vision_detection vision_pose vision_tracking)
target_compile_options(campus_bridge PRIVATE -Wall -Wextra -Wpedantic)
